cmake_minimum_required(VERSION 3.16)
project(abstraction_inference_api VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
# Generated chat@1 records, codecs and client, and the typed Chat with
# complete and stream over the shared IPC transport.
if(NOT TARGET abstraction::ipc)
 find_package(abstraction_ipc CONFIG QUIET)
endif()
if(NOT TARGET abstraction::ipc AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../abstraction-identity/cpp/CMakeLists.txt")
 add_subdirectory(../../abstraction-identity/cpp abstraction_ipc-build)
endif()
if(NOT TARGET abstraction::ipc)
 message(FATAL_ERROR "Install abstraction_ipc or provide sibling abstraction-identity. Nothing is fetched.")
endif()
add_library(abstraction_inference_api INTERFACE)
add_library(abstraction::inference_api ALIAS abstraction_inference_api)
set_target_properties(abstraction_inference_api PROPERTIES EXPORT_NAME inference_api)
target_compile_features(abstraction_inference_api INTERFACE cxx_std_17)
target_include_directories(abstraction_inference_api INTERFACE
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
 $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(abstraction_inference_api INTERFACE abstraction::ipc)
install(TARGETS abstraction_inference_api EXPORT abstraction_inference_apiTargets)
install(DIRECTORY abstraction/inference/api DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/abstraction/inference)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
set(config_dir ${CMAKE_INSTALL_LIBDIR}/cmake/abstraction_inference_api)
install(EXPORT abstraction_inference_apiTargets NAMESPACE abstraction:: DESTINATION ${config_dir})
configure_package_config_file(cmake/abstraction_inference_apiConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/abstraction_inference_apiConfig.cmake INSTALL_DESTINATION ${config_dir})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/abstraction_inference_apiConfigVersion.cmake VERSION 0.1.0 COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/abstraction_inference_apiConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/abstraction_inference_apiConfigVersion.cmake DESTINATION ${config_dir})
